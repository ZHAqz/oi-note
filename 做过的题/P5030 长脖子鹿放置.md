# 网络流建模
棋盘上的每个没有障碍的点都对应图上的一个点
如果在两个点上的长脖子鹿可以互相攻击，那么在这两个点之间连边，最后得到的图一定是一个二分图
>[!info]- 为什么
>考虑什么时候会形成一个环
>设当前点坐标为 $(x, y)$ ，那么他能攻击的坐标就是 $(x \pm 1, y \pm 3)$ ，$(x \pm 3, y \pm 1)$ 
>如果在经过 k 次攻击后坐标又回到了 $(x, y)$ ，那么 k 一定是偶数，因为如果是奇数，那么只会走到 $(x + a, y + b)$ ，其中 a ，b ，都是奇数，因为奇数个奇数相加一定还是奇数
>所以这个图中没有奇数环，所以这是一个二分图

其实到这就可以跑二分图最大匹配了，本质求得是这个图得最大独立集，用[[网络流和二分图]]中的结论，最大独立集 = 总点数 - 最大匹配数
# 但是还是有问题
连边的方向是什么？用来适配最大流求最大匹配的 s ，t 点往哪连？
我们要把这个二分图分为左部和右部，连边方向 $s \to 左部 \to 右部 \to t$
这就需要在建图的同时规定好各个边的方向，不能后面再dfs染色
# 考验注意力（套路）的地方来了
 - **从奇数行的点连向偶数行的点**
 - **从 s 连向奇数行的点**
 - **从 偶数行的点连向 t**
 上面说的所有奇偶可以反过来，就是把整个图翻转了一下
 因为互相攻击的两个长脖子鹿所在行的奇偶性一定不同，这样，就人为地把图分成了左部和右部
 P3355 骑士共存问题也是类似地套路，按 行和列的和的奇偶性分类
# 代码
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl '\n'
const int N = 205, M = 200005, inf = 2e9;

int n, m, k, b[N][N], mp[N][N], num = 0;
int dx[] = {1,3,3,1,-1,-3,-3,-1}, dy[] = {3,1,-1,-3,-3,-1,1,3};

int s, t, ans = 0;
int head[N*N], cnt = 1;
struct node {
    int to, nxt, w;
} edge[M << 1];
void addedge(int u, int v, int w) {
    edge[++cnt] = {v, head[u], w};
    head[u] = cnt;
    edge[++cnt] = {u, head[v], 0};
    head[v] = cnt;
}

int vis[N*N], dep[N*N], cur[N*N];
bool bfs() {
    for(int i = 0; i <= t; i++) vis[i] = dep[i] = 0;
    queue<int> q;
    q.push(s);
    vis[s] = dep[s] = 1;
    while(!q.empty()) {
        int u = q.front();
        q.pop();
        for(int i = head[u]; i; i = edge[i].nxt) {
            int v = edge[i].to;
            if(edge[i].w == 0 || vis[v]) continue;
            vis[v] = true;
            dep[v] = dep[u] + 1;
            q.push(v);
        }
    }
    return vis[t];
}
int dfs(int u, int flow) {
    if(u == t) return flow;
    int used = 0;
    for(int i = cur[u]; i; i = edge[i].nxt) {
        int v = edge[i].to;
        cur[u] = i;
        if(edge[i].w == 0 || dep[v] != dep[u] + 1) continue;
        int minn = dfs(v, min(flow - used, edge[i].w));
        edge[i].w -= minn;
        edge[i ^ 1].w += minn;
        used += minn;
        if(used == flow) break;
    }
    if(used == 0) dep[u] = 0;
    return used;
}

int main() {
    cin >> n >> m >> k;
    for(int i = 1; i <= k; i++) {
        int x, y;
        cin >> x >> y;
        b[x][y] = 1;
    }
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= m; j++) {
            if(b[i][j]) continue;
            mp[i][j] = ++num;
        }
    s = 0, t = num + 1;
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= m; j++) {
        	if(b[i][j]) continue;
        	if(!(i & 1)) {
        		addedge(mp[i][j], t, 1);
        		continue;
			}
			addedge(s, mp[i][j], 1);
		    for(int k = 0; k < 8; k++) {
                int nx = i + dx[k], ny = j + dy[k];
                if(nx < 1 || ny < 1 || nx > n || ny > m) continue;
                if(b[nx][ny]) continue;
                addedge(mp[i][j], mp[nx][ny], 1);
			}
		}
    while(bfs()) {
        for(int i = 0; i <= t; i++) cur[i] = head[i];
        num -= dfs(s, inf);
    }
    cout << num;
}
```
