求最小点割
与边割不同，点割指删掉一些点后，原图被分为两部分
需要用到**拆点**的技术
先把一个点拆成**入点**和**出点**，原本连接 u 和 v 的一条边现在连接了 u 的出边和 v 的入边
入点和出点之间的边权为1
两点之间的边权为 $+\infty$
这样就可以把点割转化为边割，再套最大流就行了
代码具体实现
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl '\n'
const int N = 205, M = 10005, inf = 2e9;

int n, m, s, t;
int head[N], cnt = 1;
struct node {
    int to, nxt, w;
} edge[M << 1];
void addedge(int u, int v, int w) {
    cnt++;
    edge[cnt] = {v, head[u], w};
    head[u] = cnt;
}

int dep[N], vis[N], cur[N], ans = 0;
bool bfs() {
    for(int i = 1; i <= n; i++) dep[i] = vis[i] = 0;
    queue<int> q;
    q.push(s);
    vis[s] = dep[s] = 1;
    while(!q.empty()) {
        int u = q.front();
        q.pop();
        for(int i = head[u]; i; i = edge[i].nxt) {
            int v = edge[i].to;
            if(vis[v] || edge[i].w == 0) continue;
            dep[v] = dep[u] + 1;
            vis[v] = true;
            q.push(v);
        }
    }
    return vis[t];
}
int dfs(int u, int flow) {
    if(u == t) return flow;
    int used = 0;
    for(int i = cur[u]; i; i = edge[i].nxt) {
        int v = edge[i].to;
        cur[u] = i;
        if(dep[v] != dep[u] + 1 || edge[i].w == 0) continue;
        int minn = dfs(v, min(flow - used, edge[i].w));
        edge[i].w -= minn;
        edge[i ^ 1].w += minn;
        used += minn;
        if(used == flow) break;
    }
    if(used == 0) vis[u] = 0;
    return used;
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    cin >> n >> m >> s >> t;
    for(int i = 1; i <= n; i++) {//拆点的具体实现，如果是起点或终点的话，需要把入点和出点之间的边的边权设为inf，防止删掉起点或终点
        if(i == s || i == t) {
            addedge(i, i + n, inf);
            addedge(i + n, i, 0);
        } else {
            addedge(i, i + n, 1);
            addedge(i + n, i, 0);
        }
    }
    for(int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        addedge(u + n, v, inf);
        addedge(v, u + n, 0);
        addedge(v + n, u, inf);
        addedge(u, v + n, 0);
    }
    s = s + n;
    n = n * 2;
    while(bfs()) {
        for(int i = 1; i <= n; i++) cur[i] = head[i];
        ans += dfs(s, inf);
    }
    cout << ans;
}
```
