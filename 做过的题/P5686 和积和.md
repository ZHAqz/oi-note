定义 $$s(l, r) = \sum_{i = l} ^ {r} a_{i} \times \sum_{i = l}^{r} b_{i}$$
求 $$\sum_{l = 1}^{n} \sum_{r = 1}^{n} s(l, r)$$
如果只给一个序列，，不用求乘积，那么就很好做了
这里的乘积把一次的式子转化为了二次的，无法简单地通过前面的结果推出后面的
这是就需要把和的积展开，分别计算每一项的贡献，再相加
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl '\n'
const int N = 500005;
const ll p = 1e9+7;

ll n, a[N], b[N];
ll ga[N], gb[N], f[N], ans = 0;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    cin >> n;
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= n; i++) cin >> b[i];
    for(ll i = 1; i <= n; i++) {
        ga[i] = (ga[i-1] + i * a[i] % p) % p;
        gb[i] = (gb[i-1] + i * b[i] % p) % p;
    }
    for(ll i = 1; i <= n; i++) {
    	f[i] = (f[i] + f[i-1]) % p;
    	f[i] = (f[i] + i * a[i] % p * b[i] % p) % p;
    	f[i] = (f[i] + a[i] * gb[i-1] % p) % p;
    	f[i] = (f[i] + b[i] * ga[i-1] % p) % p;
    }
	for(int i = 1; i <= n; i++)
        ans = (ans + f[i]) % p;
    cout << ans;
}
```
