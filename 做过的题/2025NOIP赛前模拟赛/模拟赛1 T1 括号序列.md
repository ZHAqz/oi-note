给一个由左右括号构成的字符串 s，对于每一个位置 i，输出有多少个子串，满足这个子串是一个合法的括号序列，并且 i 这个位置在子串中
```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long

const int N = 1e7 + 10;
const int mod = 1e9 + 7;

string s;
int a[N],b[N],match[N],n,cnt[N];
//match[i]:与位置i匹配的另一个位置，a[i]左括号所在的括号序列和左边的连续的匹配括号序列的数量和
//b[i]右括号所在的括号序列和右边的连续的匹配括号序列的数量和 
stack<int>st;

signed main()
{
    cin>>s;n=s.size();
    s=" "+s;
    for(int i=1;i<=n;i++)//求出match[i] 
        if(s[i]=='(') st.push(i);
        else
        {
            if(st.empty()) continue;
            match[i]=st.top();
            match[st.top()]=i;
            st.pop();
        }

    for(int i=1;i<=n;i++)//求出a[i] 
        if(s[i]=='(')
        {
            if(i==1||s[i-1]=='(') a[i]=1;//这个左括号的左边是左括号，左边连续的匹配括号数量为0 
            else a[i]=a[match[i-1]]+1;//s[i-1]是右括号，match[i-1]是左括号 
        }
    for(int i=n;i>=1;i--)//同理求出b[i] 
        if(s[i]==')')
        {
            if(i==n||s[i+1]==')') b[i]=1;
            else b[i]=b[match[i+1]]+1;
        }
    
 //   for(int i=1;i<=n;i++) cout<<a[i]<<" "<<b[i]<<"\n";
    for(int i=1;i<=n;i++)//统计答案，只考虑当前嵌套的括号层，cnt是一个差分数组，
//	s[i]所在的这一层上，只要在i到match[i]之间的位置，都会被贡献a[i] * b[match[i]],一个组合计数，
// 当前的括号序列和它左边的几个，右边的几个一起组成了一个新序列 
    {
        if(s[i]==')') continue;
        cnt[i]+=a[i]*b[match[i]];
        cnt[match[i]+1]-=a[i]*b[match[i]];
    }
    for(int i=1;i<=n;i++) cnt[i]+=cnt[i-1];//求前缀和就是最终的答案 
    int ans=0;
    for(int i=1;i<=n;i++)
        ans=ans+cnt[i]*i%mod;
    cout<<ans;
    return 0;
}
```
