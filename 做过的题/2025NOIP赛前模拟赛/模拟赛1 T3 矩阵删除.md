给一个 $n \times m$ 的 `01` 矩阵，我们想在每一行删除一个元素，得到一个 $n \times (m-1)$ 的矩阵。其中删除的元素的位置 $(i, a_i)$，满足 $\lvert a_i - a_{i+1} \rvert \le K$。
请问最后能得到多少种不同的矩阵。**两个矩阵如果删除的元素位置不同，但最后得到的结果相同，我们认为是相同的**。
由于答案很大，输出答案对 $10^9 + 7$ 取模的值
对于 $100 \%$ 的数据，满足 $n,m \le 3000$，$1 \le K \le m$。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl '\n'
const int N = 3005;
const ll mod = 1e9 + 7;

int n, m, k;
char a[N][N];
ll f[N][N][2], sum[N], ans;
//0去重，1所有 

int main() {
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
	cin >> n >> m >> k;
	for(int i = 1; i <= n; i++)
		for(int j = 1; j <= m; j++)
			cin >> a[i][j];
	for(int i = 1; i <= m; i++) {
		if(a[1][i] != a[1][i - 1])
			f[1][i][0] = 1;
		f[1][i][1] = 1;
		sum[i] = (sum[i - 1] + f[1][i][0]) % mod;
	} 
	for(int i = 2; i <= n; i++) {
		for(int j = 1; j <= m; j++) {
			int l = max(1, j - k);
			int r = min(m, j + k);
			f[i][j][1] = (((f[i - 1][l][1] + sum[r]) % mod - sum[l]) % mod + mod) % mod;
		}
		for(int j = 1; j <= m; j++) {
			if(a[i][j] != a[i][j - 1])
				f[i][j][0] = f[i][j][1];
			else {
				int r = min(m + 1, j + k);
				f[i][j][0] = f[i - 1][r][0];
			}
		}
		for(int j = 1; j <= m; j++)
			sum[j] = (sum[j - 1] + f[i][j][0]) % mod;
	} 
	for(int i = 1; i <= m; i++)
		ans = (ans + f[n][i][0]) % mod;
	cout << ans;
}
```
