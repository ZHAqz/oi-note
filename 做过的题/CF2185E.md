最难的就是建立这个机器人到它左边的刺的距离和这个机器人到它右边的刺的距离的映射
- 这样就能分开处理这两个距离了，可以充分利用单调性等好的性质
- 这是后面用链表或双指针的基础
其次是想到如何删除机器人
- 这里的删除直接遍历每一个死掉的机器人就行，每一个机器人最多死一次，所以复杂度是对的，不知道为什么在这个地方卡了这么久
分析到这里，这个题基本上就明了了，无论是建立两个链表，分别存排好序的到左边的距离和到右边的距离，再模拟删除机器人的过程，还是把数组排好序后用两个指针指向当前没有死的机器人，再用一个dead数组去重，都是具体的可行的方法
```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl '\n'
const int N = 200005;
const int inf = 1e9+100;

int T, n, m, k;
int a[N], b[N];
int l[N], r[N], tol[N], tor[N], nxtl[N], nxtr[N], prel[N], prer[N];
double tmp[N];
struct node {
	int l = -inf, r = inf;
} pos[N];
string com;
int live;

bool cmp(node a, node b) {
	return a.l < b.l;
}
void dell(int j) {
	nxtl[prel[j]] = nxtl[j];
	prel[nxtl[j]] = prel[j];
}
void delr(int j) {
	nxtr[prer[j]] = nxtr[j];
	prer[nxtr[j]] = prer[j];	
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
	cin >> T;
	while(T--) {
		cin >> n >> m >> k;
		for(int i = 1; i <= n; i++) cin >> a[i];
		for(int i = 1; i <= m; i++) cin >> b[i];
		sort(a + 1, a + 1 + n);
		sort(b + 1, b + 1 + m);
		live = n;
		cin >> com;
		com = " " + com;
		b[0] = -inf;
		b[m+1] = inf;
		for(int i = 1; i <= n; i++) {
			int p = lower_bound(b + 1, b + 1 + m, a[i]) - b;
			pos[i].r = b[p] - a[i];
			pos[i].l = a[i] - b[p - 1];
			r[i] = pos[i].r;
			l[i] = pos[i].l;
		}
		sort(pos + 1, pos + 1 + n, cmp);
		sort(l + 1, l + 1 + n);
		sort(r + 1, r + 1 + n);
		for(int i = 1; i <= n; i++) tmp[i] = r[i];
		for(int i = 1; i <= n; i++) {
			tor[i] = lower_bound(tmp + 1, tmp + 1 + n, pos[i].r) - tmp;
			tol[tor[i]] = i;
			tmp[tor[i]] -= 0.5;
		}
		for(int i = 0; i <= n; i++) nxtl[i] = nxtr[i] = i + 1;
		for(int i = 1; i <= n; i++) prel[i] = prer[i] = i - 1;
		int rp = 0;
		for(int i = 1; i <= k; i++) {
			if(com[i] == 'L') {
				rp--;
				for(int j = nxtl[0]; j <= n; j = nxtl[j]) {
					if(l[j] + rp > 0) break;
					live--;
					r[tor[j]] = 0;
					dell(j);
					delr(tor[j]);
				}
			} else {
				rp++;
				for(int j = nxtr[0]; j <= n; j = nxtr[j]) {
					if(r[j] - rp > 0) break;
					live--;
					l[tol[j]] = 0;
					delr(j);
					dell(tol[j]);
				}
			}
			cout << max(live, 0) << ' ';
		}
		cout << endl;
	}
}
```
