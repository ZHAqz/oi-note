tarjan求割点的时候在判断一个点是否是割点的同时会告诉你这个割点把图分成了哪几部分
```cpp
void dfs(int u, int fa) {
    dfn[u] = low[u] = ++ cnt;
    for(int v : g[u]) {
        if(v == fa) continue;
        if(!dfn[v]) {
            dfs(v, u);
            low[u] = min(low[u], low[v]);
            if(u != fa && dfn[u] <= low[v]) {
                if(dfn[b] != 0 && dfn[b] >= dfn[v])//此处割点把图分成了v的子树和除了v的子树的部分
                    cv[u] = true;
            }
        } else
            low[u] = min(low[u], dfn[v]);
    }
}
```
