和[[模拟赛1 T1 括号序列]]有异曲同工之妙，两题都有重要的一步，就是计算一个点的左边的合法区间，右边的合法区间，再引入一点组合技巧（乘法原理），从而统计出包含这个点的合法区间的总数，也算一个小trick了

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define endl '\n'
const int N =  100005;
const ll p = 1e9+7;

int n, a[N], l[N], r[N];
stack<int> st;
ll ans = 0;

int main() {
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
	cin >> n;
	for(int i = 1; i <= n; i++) {
		cin >> a[i];
	}
	for(int i = 1; i <= n; i++) {
		while(!st.empty() && a[st.top()] >= a[i]) {
			r[st.top()] = i - 1;
			st.pop();
		}
		st.push(i);
	}
	while(!st.empty()) {
		r[st.top()] = n;
		st.pop();
	}
	for(int i = n; i >= 1; i--) {
		while(!st.empty() && a[st.top()] > a[i]) {
			l[st.top()] = i + 1;
			st.pop();
		}
		st.push(i);
	}
	while(!st.empty()) {
		l[st.top()] = 1;
		st.pop();
	}
	for(int i = 1; i <= n; i++) {
		ans += 1LL * a[i] * (i - l[i] + 1) * (r[i] - i + 1) % p;
		ans %= p;
	}
	cout << ans;
}
```
