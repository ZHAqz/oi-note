# O(1)求树上两路径是否有交
先求出两个路径各自的LCA，再求出四个端点交叉组合的4个LCA
交叉组合得到的LCA中一定有两个是路径的交集的端点，具体来说，是4个LCA中深度最小的两个，记为 $x$，$y$
只有 $x$，$y$ 的深度均小于两路径各自LCA的深度时，两路径才有交集
在有交集的基础上，如果 $x \neq y$ ，说明两路径有公共边，否则说明有公共点
```cpp
bool has_common_edge(int u1, int v1, int u2, int v2) {
	int l1 = lca(u1, v1);
	int l2 = lca(u2, v2);
	int a = lca(u1, u2);
	int b = lca(u1, v2);
	int c = lca(v1, u2);
	int d = lca(v1, v2);
	int node[4] = {a, b, c, d};
	sort(node, node + 4, [](int a, int b) {
		return dep[a] > dep[b];
	});
	int x = node[0];
	int y = node[1];
	if(dep[x] < max(dep[l1], dep[l2]))
		return false;
	return x != y;
}
```

# 求树上某点的所有k级儿子
首先，求出树的bfs序，一个点的k级儿子一定在bfs序上连续
同时要求出树的dfs序，以一个点为根的子树上的点在dfs序上连续
维护 `st[i]` 表示深度 `i` 的点在bfs序上的起始位置，`ed[i]` 表示深度 `i` 的点在bfs序上的末尾位置
查询某点u的k级祖先时，在区间 $[st_{dep_{u}+k}, ed_{dep_{u}+k}]$ 上二分
具体如下
```cpp
int dep[N], bfn[N], b[N], cntbfn = 0;
int dfn[N], sz[N], cntdfn = 0;
int st[N], ed[N]; 

void dfs(int u) {
	dfn[u] = ++cntdfn;
	sz[u] = 1;
	for(int v : g[u]) {
		dfs(v);
		sz[u] += sz[v];
	}
}
void bfs() {
	queue<int> q;
	q.push(1);
	dep[1] = 1;
	while(!q.empty()) {
		int u = q.front();
		q.pop();
		b[++cnt] = u;
		bfn[u] = cntbfn;
		if(!st[dep[u]]) st[dep[u]] = cnt;
		ed[dep[u]] = cnt;
		for(int v : g[u]) {
			dep[v] = dep[u] + 1;
			q.push(v);
		}
	}
}
int getl(int u, int k) {
	int d = dep[u] + k;
	int l = st[d] - 1, r = ed[d] + 1;
	while(l + 1 < r) {
		int mid = (l + r) / 2;
		if(dfn[b[mid]] >= dfn[u]) r = mid;
		else l = mid;
	}
	return r;
}
int getr(int u, int k) {
	int d = dep[u] + k;
	int l = st[d] - 1, r = ed[d] + 1;
	while(l + 1 < r) {
		int mid = (l + r) / 2;
		if(dfn[b[mid]] < dfn[u] + sz[u]) l = mid;
		else r = mid;
	}
	return l;
}
```
